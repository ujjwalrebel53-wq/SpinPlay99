Setup (nano nahi — sirf ye commands):

  cd www
  bash setup.sh

  python3 -m venv .venv
  source .venv/bin/activate
  pip install -r requirements.txt
  playwright install chromium
  python bot.py

Seedha token + chat id dena ho to:

  bash setup.sh "BOT_TOKEN_YAHAN" "CHAT_ID_YAHAN"

  cd www

  wget -O bot_ui.py "$BASE/bot_ui.py"

  wget -O browser_session.py "$BASE/browser_session.py"
  wget -O uidai_api.py "$BASE/uidai_api.py"
  wget -O react_extract.py "$BASE/react_extract.py"
  wget -O run_all_tests.py "$BASE/run_all_tests.py"
  # v2.3 — 24h session reuse: /open MOBILE (reload skip)

  pip install -r requirements.txt
  nohup bash start.sh > bot.log 2>&1 &
