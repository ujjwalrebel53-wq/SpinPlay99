Setup (nano nahi — sirf ye commands):

  cd www
  bash setup.sh

  python3 -m venv .venv
  source .venv/bin/activate
  pip install -r requirements.txt
  playwright install chromium
  python bot.py

Seedha token + chat id dena ho to:

  bash setup.sh "BOT_TOKEN_YAHAN" "CHAT_ID_YAHAN"

VPS update:

  cd www
  BASE="https://raw.githubusercontent.com/ujjwalrebel53-wq/SpinPlay99/main/www"
  wget -O bot.py "$BASE/bot.py"
  wget -O browser_session.py "$BASE/browser_session.py"
  wget -O uidai_api.py "$BASE/uidai_api.py"
  wget -O react_extract.py "$BASE/react_extract.py"
  pkill -9 -f bot.py
  source .venv/bin/activate
  nohup python bot.py > bot.log 2>&1 &
